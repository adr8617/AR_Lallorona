function iniciarGrabacion() { let nombreSugerido = `EstudioAR_${Date.now()}`; nombreArchivoGrabado = prompt("Escribe el nombre para tu video grabado:", nombreSugerido); if (!nombreArchivoGrabado || nombreArchivoGrabado.trim() === "") { return; } if (!interfazOculta) { btnToggleInterfaz.click(); } fragmentosVideo = []; // --- SOLUCIÓN PARA UNIFICAR LA CÁMARA Y EL EFECTO EN UN SOLO VIDEO --- const canvasGrabacion = document.createElement('canvas'); canvasGrabacion.width = videoCamara.videoWidth || 1280; canvasGrabacion.height = videoCamara.videoHeight || 720; const ctxGrabacion = canvasGrabacion.getContext('2d'); // Creamos un loop interno que dibuja primero tu cámara y luego el efecto de la Llorona encima cuadro por cuadro let grabandoActivo = true; function dibujarFrameParaGrabacion() { if (!grabandoActivo) return; // 1. Dibuja la cámara de fondo ctxGrabacion.drawImage(videoCamara, 0, 0, canvasGrabacion.width, canvasGrabacion.height); // 2. Dibuja el efecto de croma (la Llorona) exactamente encima ctxGrabacion.drawImage(canvas, 0, 0, canvasGrabacion.width, canvasGrabacion.height); requestAnimationFrame(dibujarFrameParaGrabacion); } dibujarFrameParaGrabacion(); // Capturamos el stream del canvas unificado que sí contiene ambos elementos const streamCanvasUnificado = canvasGrabacion.captureStream(30); const streamCamara = videoCamara.srcObject; const tracksCombinados = [...streamCanvasUnificado.getVideoTracks()]; if (videoMultimedia.srcObject && videoMultimedia.srcObject.getAudioTracks().length > 0) { tracksCombinados.push(...videoMultimedia.srcObject.getAudioTracks()); } const streamFinal = new MediaStream(tracksCombinados); try { mediaRecorder = new MediaRecorder(streamFinal, { mimeType: 'video/webm; codecs=vp9' }); } catch (e) { mediaRecorder = new MediaRecorder(streamFinal); } mediaRecorder.ondataavailable = (event) => { if (event.data && event.data.size > 0) fragmentosVideo.push(event.data); }; mediaRecorder.onstop = () => { grabandoActivo = false; // Detiene el loop de unificación const blob = new Blob(fragmentosVideo, { type: 'video/webm' }); const urlBlob = URL.createObjectURL(blob); const a = document.createElement('a'); a.style.display = 'none'; a.href = urlBlob; a.download = `${nombreArchivoGrabado.replace(/[^a-z0-9_]/gi, '_')}.webm`; document.body.appendChild(a); a.click(); setTimeout(() => { document.body.removeChild(a); window.URL.revokeObjectURL(urlBlob); alert(`¡Video guardado con éxito con la Llorona incluida!`); }, 100); if (interfazOculta) { btnToggleInterfaz.click(); } }; mediaRecorder.start(); estaGrabando = true; btnGrabar.classList.add('grabando'); }
